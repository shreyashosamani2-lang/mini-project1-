<script>
(function() {
  const wrapper   = document.getElementById('heroSlider');
  const track     = document.getElementById('sliderTrack');
  const dots      = Array.from(document.querySelectorAll('.hero-slider__dot'));
  const progress  = document.getElementById('sliderProgress');
  const swipeHint = document.getElementById('swipeHint');
  const imgs      = Array.from(document.querySelectorAll('.hero-slider__model-img'));
  const total     = dots.length;

  const AUTOPLAY_DELAY = {{ section.settings.autoplay_speed | default: 4000 }};
  const SWIPE_THRESHOLD = 40;

  let current = 0;
  let autoplayTimer = null;

  // ── SET IMAGE HEIGHTS based on screen size ────────────────
  function applyImageSizes() {
    const isMobile = window.innerWidth <= 768;
    imgs.forEach(img => {
      const pct = isMobile
        ? parseInt(img.dataset.mobile  || 75)
        : parseInt(img.dataset.desktop || 90);
      img.style.height = pct + 'vh';
    });
  }

  applyImageSizes();
  window.addEventListener('resize', applyImageSizes);

  // ── SLIDE TO INDEX ────────────────────────────────────────
  function goTo(index, animate) {
    const prev = current;
    current = (index + total) % total;

    // Move track: each slot is 100vw
    if (animate === false) {
      track.style.transition = 'none';
    } else {
      track.style.transition = 'transform 0.7s cubic-bezier(0.77, 0, 0.18, 1)';
    }
    track.style.transform = 'translateX(-' + (current * 100) + 'vw)';

    dots[prev].classList.remove('is-active');
    dots[current].classList.add('is-active');

    resetProgress();
  }

  function resetProgress() {
    progress.classList.remove('animating');
    progress.style.width = '0%';
    requestAnimationFrame(() => requestAnimationFrame(() => {
      progress.style.transitionDuration = AUTOPLAY_DELAY + 'ms';
      progress.classList.add('animating');
      progress.style.width = '100%';
    }));
  }

  function startAutoplay() {
    clearInterval(autoplayTimer);
    autoplayTimer = setInterval(() => goTo(current + 1), AUTOPLAY_DELAY);
  }

  function stopAutoplay() {
    clearInterval(autoplayTimer);
    progress.classList.remove('animating');
  }

  // Dot clicks
  dots.forEach(dot => {
    dot.addEventListener('click', () => {
      goTo(parseInt(dot.dataset.index));
      stopAutoplay(); startAutoplay();
    });
  });

  // ── DRAG / SWIPE ──────────────────────────────────────────
  let dragStartX = 0;
  let dragCurrentX = 0;
  let dragging = false;
  let hasMoved = false;

  function onDragStart(x) {
    dragStartX = x;
    dragCurrentX = x;
    dragging = true;
    hasMoved = false;
    wrapper.classList.add('is-dragging');
    stopAutoplay();
    // Disable transition while dragging for live follow
    track.style.transition = 'none';
  }

  function onDragMove(x) {
    if (!dragging) return;
    dragCurrentX = x;
    const diff = dragStartX - x;
    if (Math.abs(diff) > 5) hasMoved = true;
    // Live drag: move track with finger/cursor
    const basePx = current * window.innerWidth;
    track.style.transform = 'translateX(-' + (basePx + diff) + 'px)';
  }

  function onDragEnd(x) {
    if (!dragging) return;
    dragging = false;
    wrapper.classList.remove('is-dragging');

    const dx = dragStartX - x;
    const dy = 0; // not tracking Y here, just horizontal

    if (hasMoved && Math.abs(dx) > SWIPE_THRESHOLD) {
      goTo(dx > 0 ? current + 1 : current - 1);
      swipeHint.classList.add('hide');
    } else {
      // Snap back
      goTo(current);
    }
    startAutoplay();
  }

  // Mouse
  wrapper.addEventListener('mousedown', e => {
    if (e.target.closest('a, button')) return;
    e.preventDefault();
    onDragStart(e.clientX);
  });

  document.addEventListener('mousemove', e => {
    if (!dragging) return;
    onDragMove(e.clientX);
  });

  document.addEventListener('mouseup', e => {
    if (!dragging) return;
    onDragEnd(e.clientX);
  });

  // Touch
  wrapper.addEventListener('touchstart', e => {
    if (e.target.closest('a, button')) return;
    onDragStart(e.touches[0].clientX);
  }, { passive: true });

  wrapper.addEventListener('touchmove', e => {
    if (!dragging) return;
    onDragMove(e.touches[0].clientX);
  }, { passive: true });

  wrapper.addEventListener('touchend', e => {
    if (!dragging) return;
    onDragEnd(e.changedTouches[0].clientX);
  });

  // Pause on hover (desktop only)
  wrapper.addEventListener('mouseenter', () => { if (!dragging) stopAutoplay(); });
  wrapper.addEventListener('mouseleave', () => { if (!dragging) startAutoplay(); });

  // ── INIT ─────────────────────────────────────────────────
  track.style.width = (total * 100) + 'vw';
  goTo(0, false);
  startAutoplay();
  setTimeout(() => swipeHint.classList.add('hide'), 3500);
})();
</script>
